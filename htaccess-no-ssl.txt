# Temporary .htaccess for BEFORE AutoSSL has issued a certificate.
# Rename to .htaccess, then swap back to the SSL version once https works.
# Note: the microphone, the service worker and Add to Home Screen all require https.

# Ellipsis — Hostinger (LiteSpeed) config, pre-SSL
Options -Indexes
DirectoryIndex index.html
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
  RewriteEngine On
  # SSL redirect intentionally disabled — enable it once AutoSSL has issued a certificate
  # pretty paths: /app -> /app/
  RewriteCond %{REQUEST_FILENAME} -d
  RewriteRule ^(.+[^/])$ /$1/ [L,R=301]
</IfModule>

<IfModule mod_mime.c>
  AddType application/manifest+json      .webmanifest
  AddType text/javascript                .js .jsx
  AddType audio/wav                      .wav
  AddType image/png                      .png
</IfModule>

<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/javascript application/javascript application/json application/manifest+json image/svg+xml
</IfModule>

<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresDefault                          "access plus 1 hour"
  ExpiresByType text/html                 "access plus 0 seconds"
  ExpiresByType text/javascript           "access plus 7 days"
  ExpiresByType audio/wav                 "access plus 1 year"
  ExpiresByType image/png                 "access plus 1 year"
</IfModule>

# never cache the workers or the app documents
<FilesMatch "^(sw\.js|index\.html|moderation\.html)$">
  Header set Cache-Control "no-cache, must-revalidate"
</FilesMatch>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "microphone=(self), camera=(), geolocation=()"
  Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' 'unsafe-eval' https://unpkg.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com; img-src 'self' data: blob:; media-src 'self' blob:; connect-src 'self' https://unpkg.com; frame-ancestors 'self'"
</IfModule>
